function Eyebrow({ children }: { children: React.ReactNode }) {
  return <div style={{ color: C.brightGold, font: "800 12px Arial", letterSpacing: ".13em" }}>{children}</div>;
}
function Pill({ children, background, color, uppercase = false }: { children: React.ReactNode; background: string; color: string; uppercase?: boolean }) {
  return <span style={{ background, color, borderRadius: 9999, padding: "5px 8px", font: `800 12px Arial`, letterSpacing: uppercase ? ".05em" : undefined }}>{children}</span>;
}
function Avatar({ initials, size = 33 }: { initials: string; size?: number }) {
  return <div style={{ width: size, height: size, flex: `0 0 ${size}px`, borderRadius: 9999, background: C.navySoft, color: C.midnight, display: "grid", placeItems: "center", font: `700 ${size < 32 ? 11 : 12}px Arial` }}>{initials}</div>;
}
function Card({ children, style }: { children: React.ReactNode; style?: React.CSSProperties }) {
  return <section style={{ background: C.paper, border: `1px solid ${C.line}`, borderRadius: 14, padding: 20, ...style }}>{children}</section>;
}
function KpiCard({ label, number, trendValue, trendSuffix, trendDir, trendColor }: { label: string; number: string; trendValue: string; trendSuffix: string; trendDir: "up" | "down"; trendColor: string }) {
  const TrendIcon = trendDir === "up" ? ArrowUp : ArrowDown;
  return <Card style={{ display: "flex", flexDirection: "column", gap: 0, minHeight: 132, padding: 10 }}>
    <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", padding: "6px 8px 10px" }}>

    <div style={{ flex: 1, background: "#fff", border: `1px solid ${C.line}`, borderRadius: 12, padding: "14px 14px 14px" }}>
      <div style={{ color: C.ink, font: "700 28px Georgia", letterSpacing: "-.02em", lineHeight: 1 }}>{number}</div>
      <div style={{ display: "flex", alignItems: "center", gap: 5, marginTop: 12 }}>
        <TrendIcon size={14} color={trendColor} strokeWidth={2.5} />
        <span style={{ color: trendColor, font: "700 13px Arial" }}>{trendValue}</span>
        <span style={{ color: C.slate, font: "13px Arial" }}>{trendSuffix}</span>

}
